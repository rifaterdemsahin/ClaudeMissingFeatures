Run it

cd ~/projects/ClaudeMissingFeatures

# 1. Remotion Studio (first time: npm install)
cd remotion
npm install
ln ~/Downloads/claude-missing-features-1080p.mp4 public/base.mp4   # hard link, once (a symlink breaks renders)
npm run studio                                    # open http://localhost:3000/LongForm
npm run render                                    # final video -> remotion/out/long.mp4
npm run still                                     # thumbnail  -> remotion/out/thumbnail.png
npx tsc --noEmit                                  # typecheck

# 2. Site and player (new terminal)
cd ~/projects/ClaudeMissingFeatures
scripts/serve.sh                                  # http://localhost:8000/player.html

# 3. Open in Chrome
open -a "Google Chrome" http://localhost:3000/LongForm
open -a "Google Chrome" http://localhost:8000/player.html
